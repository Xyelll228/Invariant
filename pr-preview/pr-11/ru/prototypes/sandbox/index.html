Проверка изоляции

Вставьте пример и запустите: код попробует прочитать хранилище приложения, отправить запрос в сеть или уйти в бесконечный цикл. Во время бесконечного цикла страница продолжает работать — например, переключается тема.

Вставить пример:
// Пробуем открыть базу приложения «invariant», где хранится прогресс.
function probe() {
  return new Promise((resolve) => {
    let request;
    try {
      request = indexedDB.open('invariant');
    } catch (error) {
      resolve('IndexedDB приложения: нет доступа (' + error.name + ')');
      return;
    }
    request.onerror = (event) => {
      event.preventDefault();
      resolve('IndexedDB приложения: нет доступа (' + request.error.name + ')');
    };
    request.onsuccess = () => {
      const stores = Array.from(request.result.objectStoreNames);
      request.result.close();
      resolve(
        stores.length === 0
          ? 'IndexedDB: открылась пустая база песочницы, данных приложения в ней нет'
          : 'IndexedDB приложения: видны хранилища ' + JSON.stringify(stores),
      );
    };
  });
}

Esc, затем Tab — выйти из редактора. Ctrl+Z — отменить.

Как это устроено

  • Код выполняется во вложенном окне с атрибутом sandbox без allow-same-origin. У такого окна непрозрачный источник: хранилище приложения ему недоступно.
  • Внутри окна — фоновый поток. По истечении лимита времени поток уничтожается, а страница остаётся рабочей.
  • Политика безопасности контента (CSP) окна запрещает сетевые запросы, картинки, стили и вложенные окна.
